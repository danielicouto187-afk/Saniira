import math
import time

class BobinaDeCoerencia:
    def __init__(self):
        self.frequencia = 7.77
        self.torque = 7
    
    def girar(self, segundos=3):
        print(f"🌀 Bobina girando a {self.frequencia} Hz...")

            print(f"Volta {t+1}: campo = {campo:.4f}")

        print("✅ Bobina de coerência estabelecida.")
